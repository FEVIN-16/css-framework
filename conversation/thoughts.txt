Step 1 — Create the Project

Create the main project folder.

mkdir framify
cd framify
npm init -y

Install base dependencies:

npm install typescript ts-node @types/node

Install core tools:

npm install lightningcss fast-glob chokidar commander

Dev tools:

npm install -D tsup

tsup will bundle your library.

Step 2 — Initialize TypeScript

Run:

npx tsc --init

Modify tsconfig.json:

{
  "compilerOptions": {
    "target": "ES2022",
    "module": "ESNext",
    "moduleResolution": "Node",
    "outDir": "dist",
    "rootDir": "src",
    "strict": true,
    "declaration": true
  }
}

Step 6 — Create File Scanner

Create:

src/scanner/scanFiles.ts

Example:

import fg from "fast-glob"
import fs from "fs"

export async function scanFiles() {
  const files = await fg(["src/**/*.{html,js,ts,jsx,tsx}"])

  const classes = new Set<string>()

  for (const file of files) {
    const content = fs.readFileSync(file, "utf-8")

    const matches = content.match(/class=["'`](.*?)["'`]/g)

    if (matches) {
      matches.forEach(match => {
        const cls = match
          .replace(/class=["'`]/, "")
          .replace(/["'`]/, "")
          .split(" ")

        cls.forEach(c => classes.add(c))
      })
    }
  }

  return [...classes]
}

This extracts:

p-4
flex
bg-blue-500
Step 7 — Create Utility Generator

Create:

src/generator/generateUtilities.ts

Example:

export function generateUtilities(classes: string[]) {
  let css = ""

  classes.forEach(cls => {
    if (cls.startsWith("p-")) {
      const value = cls.split("-")[1]

      css += `
.${cls} {
  padding: ${value}px;
}
`
    }
  })

  return css
}
Step 8 — Process CSS with LightningCSS

Create:

src/css/processCss.ts

Example:

import { transform } from "lightningcss"

export function processCss(css: string) {
  const result = transform({
    filename: "framify.css",
    code: Buffer.from(css),
    minify: true
  })

  return result.code.toString()
}
Step 9 — Create Watch Mode

Create:

src/watcher/watchFiles.ts

Example:

import chokidar from "chokidar"

export function watchFiles(callback: () => void) {
  const watcher = chokidar.watch("src")

  watcher.on("change", () => {
    console.log("File changed, rebuilding...")
    callback()
  })
}

export function generateColorUtilities(colors: any) {
  let css = ""

  for (const name in colors) {
    css += `.bg-${name} { background:${colors[name]}; }`
  }

  return css
}

Step 13 — Main Framework Entry

Create:

src/index.ts

Example:

import { scanFiles } from "./scanner/scanFiles"
import { generateUtilities } from "./generator/generateUtilities"
import { processCss } from "./css/processCss"

export async function build() {
  const classes = await scanFiles()

  const css = generateUtilities(classes)

  const finalCss = processCss(css)

  console.log(finalCss)
}
